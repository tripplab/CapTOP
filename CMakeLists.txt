cmake_minimum_required(VERSION 3.16)
project(captop LANGUAGES CXX)
option(CAPTOP_WITH_GUDHI "Build with GUDHI support" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(captop captop.cpp)
target_compile_options(captop PRIVATE -Wall -Wextra -pedantic)
if(CAPTOP_WITH_GUDHI)
  find_path(GUDHI_INCLUDE_DIR gudhi/Bitmap_cubical_complex.h HINTS ${GUDHI_INCLUDE_DIR})
  if(NOT GUDHI_INCLUDE_DIR)
    message(FATAL_ERROR "GUDHI headers not found. Set -DGUDHI_INCLUDE_DIR=/path/to/gudhi/include or configure with -DCAPTOP_WITH_GUDHI=OFF.")
  endif()
  message(STATUS "CAPTOP_WITH_GUDHI: ON")
  message(STATUS "GUDHI include path: ${GUDHI_INCLUDE_DIR}")
  message(STATUS "GUDHI support: enabled")
  message(STATUS "Stage 5 betti command: GUDHI authoritative + fallback cross-checks")
  target_include_directories(captop PRIVATE ${GUDHI_INCLUDE_DIR})
  target_compile_definitions(captop PRIVATE CAPTOP_WITH_GUDHI=1)
else()
  message(STATUS "CAPTOP_WITH_GUDHI: OFF")
  message(STATUS "GUDHI support: disabled")
  message(STATUS "Stage 5 betti command: fallback diagnostics only")
endif()
